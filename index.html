from abc import ABC, abstractmethod from datetime import datetime import json import hashlib import uuid # ========================= # MODELS # ========================= class User: def __init__(self, username, password, role="user"): self.id = str(uuid.uuid4()) self.username = username self.password_hash = self._hash_password(password) self.role = role self.created_at = datetime.now() def _hash_password(self, password): return hashlib.sha256(password.encode()).hexdigest() def verify_password(self, password): return self.password_hash == hashlib.sha256( password.encode() ).hexdigest() class Record: def __init__(self, title, value): self.id = str(uuid.uuid4()) self.title = title self.value = value self.created_at = datetime.now() def to_dict(self): return { "id": self.id, "title": self.title, "value": self.value, "created_at": self.created_at.isoformat() } # ========================= # REPOSITORY # ========================= class Repository(ABC): @abstractmethod def save(self, item): pass @abstractmethod def get_all(self): pass @abstractmethod def delete(self, item_id): pass class RecordRepository(Repository): def __init__(self): self.records = [] def save(self, record): self.records.append(record) def get_all(self): return self.records def delete(self, item_id): self.records = [ r for r in self.records if r.id != item_id ] # ========================= # AUDIT SYSTEM # ========================= class AuditLogger: @staticmethod def log(action, username): timestamp = datetime.now() with open("audit.log", "a") as file: file.write( f"{timestamp} | " f"{username} | " f"{action}\n" ) # ========================= # BUSINESS LOGIC # ========================= class RecordService: def __init__(self, repository): self.repository = repository def create_record(self, title, value): if len(title) < 3: raise ValueError( "Title must be at least 3 characters" ) if value < 0: raise ValueError( "Value cannot be negative" ) record = Record(title, value) self.repository.save(record) return record def calculate_statistics(self): records = self.repository.get_all() if not records: return None values = [r.value for r in records] return { "count": len(values), "total": sum(values), "average": sum(values) / len(values), "max": max(values), "min": min(values) } # ========================= # REPORT ENGINE # ========================= class ReportGenerator: def generate_json_report(self, records): report = [ record.to_dict() for record in records ] with open( "report.json", "w" ) as file: json.dump( report, file, indent=4 ) # ========================= # APPLICATION # ========================= class Application: def __init__(self): self.users = {} self.repo = RecordRepository() self.service = RecordService( self.repo ) self.report_generator = ReportGenerator() def register(self, username, password, role="user"): if username in self.users: raise Exception( "Username already exists" ) user = User( username, password, role ) self.users[username] = user AuditLogger.log( "REGISTER", username ) def login( self, username, password ): user = self.users.get(username) if not user: return None if user.verify_password(password): AuditLogger.log( "LOGIN", username ) return user return None def add_record( self, user, title, value ): record = self.service.create_record( title, value ) AuditLogger.log( f"ADD_RECORD:{record.id}", user.username ) return record # ========================= # MAIN # ========================= if __name__ == "__main__": app = Application()
